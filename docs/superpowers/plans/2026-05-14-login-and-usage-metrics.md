# Login and Usage Metrics Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a per-user roll-up of login + usage counters in Firestore, surfaced in a new Admin page section, so Matt can see who is using songpicker and at a high level what they're doing.

**Architecture:** A new server module `functions/src/metrics.ts` exposes `recordMetric(uid, email, action)` that bumps a counter doc at `metrics/{uid}`. The SPA fires a dedicated `recordSignIn` callable once on auth state change; four existing callables (`getPlanContext`, `suggestSongs`, `placeSongInSlot`, `clearSongFromSlot`) gain one fire-and-forget call to `recordMetric`. A new admin-only `getMetrics` callable powers a `MetricsSection` on the Admin page. The Firestore-write shape is generated by a pure helper `buildMetricUpdate` for unit-test coverage; the I/O wrapper stays untested, consistent with how `sessions.ts` is treated.

**Tech Stack:** TypeScript, Firebase Functions (Gen 2, Node 22), firebase-admin Firestore, React 18 + Vite, Vitest. Spec at `docs/superpowers/specs/2026-05-14-login-and-usage-metrics-design.md`.

```typescript
import { FieldValue, Timestamp, getFirestore } from 'firebase-admin/firestore'

export const METRIC_ACTIONS = [
  'signIns',
  'plansOpened',
  'aiSuggestions',
  'slotCommits',
  'slotClears',
] as const

export type MetricAction = (typeof METRIC_ACTIONS)[number]

export interface StoredMetrics {
  email?: string
  firstSeenAt?: Timestamp
  lastSeenAt?: Timestamp
  signIns?: number
  plansOpened?: number
  aiSuggestions?: number
  slotCommits?: number
  slotClears?: number
}

// Pure helper — returns the Firestore merge payload for one bump.
// `existing` is null when the doc doesn't exist yet; that's when we
// set `firstSeenAt`. Everything else (email mirror, lastSeenAt,
// counter increment) happens on every write.
export function buildMetricUpdate(
  existing: StoredMetrics | null,
  action: MetricAction,
  email: string,
): Record<string, unknown> {
  const update: Record<string, unknown> = {
    email,
    lastSeenAt: FieldValue.serverTimestamp(),
    [action]: FieldValue.increment(1),
  }
  if (!existing) {
    update.firstSeenAt = FieldValue.serverTimestamp()
  }
  return update
}
```

```typescript
// View shape returned to the Admin SPA. Timestamps are serialized to
// ISO strings because callable responses can't carry Timestamp objects.
export interface UserMetricsView {
  uid: string
  email: string
  firstSeenAt: string | null
  lastSeenAt: string | null
  signIns: number
  plansOpened: number
  aiSuggestions: number
  slotCommits: number
  slotClears: number
}

// Fire-and-forget on the calling side. Wrapped in try/catch so a
// flaky Firestore write never propagates into the user-facing
// callable. Logs a warning so failures aren't silent in function
// logs.
export async function recordMetric(
  uid: string,
  email: string,
  action: MetricAction,
): Promise<void> {
  try {
    const db = getFirestore()
    const ref = db.collection('metrics').doc(uid)
    const snap = await ref.get()
    const existing = snap.exists ? (snap.data() as StoredMetrics) : null
    await ref.set(buildMetricUpdate(existing, action, email), { merge: true })
  } catch (err) {
    console.warn('[metrics] write failed:', err)
  }
}

// Admin-page read. Sorted by lastSeenAt descending so the most
// recently active user is at the top. Missing fields default to
// 0 / null so the SPA can render a uniform table.
export async function loadAllMetrics(): Promise<UserMetricsView[]> {
  const db = getFirestore()
  const snap = await db.collection('metrics').get()
  const rows: UserMetricsView[] = snap.docs.map((doc) => {
    const data = doc.data() as StoredMetrics
    return {
      uid: doc.id,
      email: data.email ?? '(unknown)',
      firstSeenAt: data.firstSeenAt?.toDate().toISOString() ?? null,
      lastSeenAt: data.lastSeenAt?.toDate().toISOString() ?? null,
      signIns: data.signIns ?? 0,
      plansOpened: data.plansOpened ?? 0,
      aiSuggestions: data.aiSuggestions ?? 0,
      slotCommits: data.slotCommits ?? 0,
      slotClears: data.slotClears ?? 0,
    }
  })
  rows.sort((a, b) => {
    if (a.lastSeenAt === b.lastSeenAt) return 0
    if (a.lastSeenAt === null) return 1
    if (b.lastSeenAt === null) return -1
    return a.lastSeenAt < b.lastSeenAt ? 1 : -1
  })
  return rows
}
```

```typescript
// Fired by the SPA on every Firebase Auth transition to signed-in.
// Counts as a fresh sign-in. requireAllowed runs first so the
// metric is never written for a denied caller. Returns `{ ok: true }`
// to keep the SPA's existing call/await pattern uniform.
export const recordSignIn = onCall(async (req) => {
```

```typescript
function MetricsSection() {
  const [users, setUsers] = useState<UserMetricsView[] | null>(null)
  const [error, setError] = useState<string | null>(null)

  // Mount fetch via .then() to keep setState off the synchronous effect
  // path (react-hooks/set-state-in-effect) — same shape AccessSection uses.
  useEffect(() => {
    let cancelled = false
    getMetrics()
      .then((res) => {
        if (!cancelled) setUsers(res.data.users)
      })
      .catch((e) => {
        if (!cancelled) setError((e as Error).message)
      })
    return () => {
      cancelled = true
    }
  }, [])

  return (
    <section className="admin-section">
      <h3>Usage</h3>
      {error && <div className="error">{error}</div>}
      {users === null ? (
        <p className="muted">Loading…</p>
      ) : users.length === 0 ? (
        <p className="muted">No activity recorded yet.</p>
      ) : (
        <table className="metrics-table">
          <thead>
            <tr>
              <th>Email</th>
              <th>Sign-ins</th>
              <th>Plans opened</th>
              <th>AI suggestions</th>
              <th>Commits</th>
              <th>Clears</th>
              <th>First seen</th>
              <th>Last seen</th>
            </tr>
          </thead>
          <tbody>
            {users.map((u) => (
              <tr key={u.uid}>
                <td>{u.email}</td>
                <td>{u.signIns}</td>
                <td>{u.plansOpened}</td>
                <td>{u.aiSuggestions}</td>
                <td>{u.slotCommits}</td>
                <td>{u.slotClears}</td>
                <td title={u.firstSeenAt ?? ''}>{formatRelative(u.firstSeenAt)}</td>
                <td title={u.lastSeenAt ?? ''}>{formatRelative(u.lastSeenAt)}</td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
    </section>
  )
}

// Compact relative-time formatter for the metrics table. Returns
// short forms ("2h ago", "3d ago") for recent times and a date
// stamp for anything older than ~30 days. ISO input → display string.
function formatRelative(iso: string | null): string {
  if (!iso) return '—'
  const then = new Date(iso).getTime()
  if (!Number.isFinite(then)) return iso
  const diffMs = Date.now() - then
  if (diffMs < 0) return 'just now'
  const sec = Math.floor(diffMs / 1000)
  if (sec < 60) return 'just now'
  const min = Math.floor(sec / 60)
  if (min < 60) return `${min}m ago`
  const hr = Math.floor(min / 60)
  if (hr < 24) return `${hr}h ago`
  const day = Math.floor(hr / 24)
  if (day < 30) return `${day}d ago`
  return new Date(iso).toISOString().slice(0, 10)
}
```
